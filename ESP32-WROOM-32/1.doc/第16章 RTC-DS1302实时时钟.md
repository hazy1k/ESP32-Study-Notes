# 第十六章 DS1302 实时时钟

## 1. 导入

DS1302 是一款“3 线串行”（CLK/IO/CE）的低功耗 RTC，带后备电池 VBAT，可在断电期间保持走时。它不是标准 SPI/I2C，而是自有半双工协议（单根 IO 双向）。常见小板自带 CR2032 电池座，成本低、易上手。

本章给出 DS1302 的可靠接线建议，以及完整的 MicroPython 驱动与示例：读取/设置时间、启动/停止时钟、禁用涓流充电（保护一次性电池）、与系统 RTC/NTP 的联动。

## 2. 硬件设计

- 供电与电平：
  - DS1302 工作电压 2.0–5.5V。若以 5V 供电，IO 引脚的高电平也会接近 5V，可能损坏 3.3V 的 ESP32。
  - 建议整板用 3.3V 供电（VCC=3.3V），这样 IO 高电平≈3.3V，直接安全连 ESP32。
- 接线（推荐 3.3V 版本小板）：

```c
ESP32 3V3  -> DS1302 VCC
ESP32 GND  -> DS1302 GND
ESP32 GPIO22 -> DS1302 CLK   // 任意可用 GPIO
ESP32 GPIO21 -> DS1302 IO    // 单线双向数据
ESP32 GPIO5  -> DS1302 CE    // 片选（亦称 RST）
DS1302 VBAT  -> CR2032 电池正极（负极接 GND）
```

- 注意涓流充电（Trickle Charger）：
  - 许多 DS1302 模块连着不可充电的 CR2032。务必在软件中“关闭涓流充电寄存器”，避免对不可充电电池反向充电。
  - 
- 走线与旁路：
  - VCC 近端并 0.1µF 去耦；线尽量短；IO 线不必加上拉（片内/板上已有）。

## 3. 协议与寄存器简述

- 3 线时序：CE 置高开始一次传输；时钟沿驱动数据移位；IO 为 LSB-first。
- 读/写命令：先发“寄存器地址字节”，LSB=1 表示读，LSB=0 表示写；随后传输数据字节。
- 主要寄存器（8 位 BCD）：
  - 0x80 秒（bit7=CH，1=停止；需清零使能）
  - 0x82 分
  - 0x84 时（bit7=12/24h，0=24h）
  - 0x86 日（1..31）
  - 0x88 月（1..12）
  - 0x8A 周（1..7）
  - 0x8C 年（00..99，表示 2000..2099）
  - 0x8E 写保护（bit7=1 禁止写）
  - 0x90 涓流充电控制（写 0x00 关闭）

## 4. 驱动封装 ds1302.py（完整可用）

```python
# 文件：ds1302.py
from machine import Pin
import time

# 地址常量（LSB 在命令阶段用于读写选择）
_REG_SEC   = 0x80
_REG_MIN   = 0x82
_REG_HOUR  = 0x84
_REG_DATE  = 0x86
_REG_MONTH = 0x88
_REG_WDAY  = 0x8A
_REG_YEAR  = 0x8C
_REG_WP    = 0x8E
_REG_TCR   = 0x90

def _bcd2i(x): return (x >> 4) * 10 + (x & 0x0F)
def _i2bcd(x): return ((x // 10) << 4) | (x % 10)

class DS1302:
    """
    DS1302 三线接口驱动（LSB-first）
    接线：CLK, IO(双向), CE
    """
    def __init__(self, clk, io, ce):
        self.clk = Pin(clk, Pin.OUT, value=0)
        self.io  = Pin(io,  Pin.OUT, value=0)
        self.ce  = Pin(ce,  Pin.OUT, value=0)
        # 安全默认：关闭涓流充电
        self.disable_trickle()
        # 取消写保护（便于设置时间；若只读可再打开）
        self.write_protect(False)

    def _start(self):
        self.ce.value(1)
        # 时钟空闲低
        self.clk.value(0)

    def _end(self):
        self.ce.value(0)

    def _write_byte(self, b):
        # LSB-first
        self.io.init(Pin.OUT)
        for i in range(8):
            self.io.value((b >> i) & 1)
            self.clk.value(1)
            self.clk.value(0)

    def _read_byte(self):
        val = 0
        self.io.init(Pin.IN)
        for i in range(8):
            bit = self.io.value()
            val |= (bit << i)
            self.clk.value(1)
            self.clk.value(0)
        return val

    def _read_reg(self, reg):
        self._start()
        self._write_byte(reg | 0x01)  # 读：LSB=1
        val = self._read_byte()
        self._end()
        return val

    def _write_reg(self, reg, val):
        self._start()
        self._write_byte(reg & 0xFE)  # 写：LSB=0
        self._write_byte(val & 0xFF)
        self._end()

    def write_protect(self, enable=True):
        self._write_reg(_REG_WP, 0x80 if enable else 0x00)

    def disable_trickle(self):
        # 寄存器写 0 关闭涓流，保护 CR2032
        self._write_reg(_REG_TCR, 0x00)

    def halt(self, stop=True):
        """
        置位/清除 秒寄存器 CH 位（1=停止振荡；0=运行）
        """
        sec = self._read_reg(_REG_SEC)
        if stop:
            sec |= 0x80
        else:
            sec &= 0x7F
        self._write_reg(_REG_SEC, sec)

    def datetime(self):
        """
        读取日期时间（24h）：返回 (year, month, day, hour, minute, second, weekday)
        weekday: 1..7
        """
        ss = self._read_reg(_REG_SEC) & 0x7F
        mm = self._read_reg(_REG_MIN)
        hh = self._read_reg(_REG_HOUR)
        dd = self._read_reg(_REG_DATE)
        mo = self._read_reg(_REG_MONTH)
        wd = self._read_reg(_REG_WDAY)
        yy = self._read_reg(_REG_YEAR)

        # 24 小时制（bit7=0）
        hh = _bcd2i(hh & 0x3F)
        return (2000 + _bcd2i(yy),
                _bcd2i(mo), _bcd2i(dd),
                hh, _bcd2i(mm), _bcd2i(ss),
                _bcd2i(wd))

    def set_datetime(self, y, mo, d, hh, mm, ss, wday=None):
        """
        设置日期时间（24h）
        wday: 1..7（可选，None 则自动计算）
        """
        # 自动计算星期（0=Mon..6），转 1..7
        if wday is None:
            secs = time.mktime((y, mo, d, hh, mm, ss, 0, 0))
            wday = time.localtime(secs)[6] + 1
            if wday == 8:
                wday = 1

        # 允许写
        self.write_protect(False)
        # 清 CH & 写秒
        self._write_reg(_REG_SEC, _i2bcd(ss) & 0x7F)
        self._write_reg(_REG_MIN, _i2bcd(mm))
        # 24h：bit7=0
        self._write_reg(_REG_HOUR, _i2bcd(hh) & 0x3F)
        self._write_reg(_REG_DATE, _i2bcd(d))
        self._write_reg(_REG_MONTH, _i2bcd(mo))
        self._write_reg(_REG_WDAY, _i2bcd(wday))
        self._write_reg(_REG_YEAR, _i2bcd((y - 2000) % 100))
        # 可选：重新打开写保护
        # self.write_protect(True)
```

## 5. 快速上手：读写时间

```python
# 文件：main_ds1302.py
from ds1302 import DS1302
from machine import Pin
import time

# 接线示例（见第 2 节）
CLK, IO, CE = 22, 21, 5
rtc = DS1302(CLK, IO, CE)

# 第一次使用可设置时间（设置后注释掉）
# rtc.set_datetime(2025, 1, 1, 12, 0, 0)

# 确保运行（清除 CH）
rtc.halt(False)

while True:
    y, mo, d, hh, mm, ss, wd = rtc.datetime()
    print("{:04d}-{:02d}-{:02d} {:02d}:{:02d}:{:02d} (W{})".format(y,mo,d,hh,mm,ss,wd))
    time.sleep(1)
```

## 6. 与系统 RTC/NTP 联动

- 上电流程建议：先从 DS1302 读取时间设置系统 RTC；若随后联网成功，再用 NTP 校时，并把新时间回写 DS1302。
- 注意：NTP 返回 UTC；显示本地时区需加偏移。

示例：

```python
# 文件：rtc_sync_ds1302.py
from ds1302 import DS1302
from machine import Pin, RTC
import time, network, ntptime

CLK, IO, CE = 22, 21, 5
chip = DS1302(CLK, IO, CE)

def set_system_rtc(y, mo, d, hh, mm, ss):
    rtc = RTC()
    secs = time.mktime((y,mo,d,hh,mm,ss,0,0))
    wday = time.localtime(secs)[6]
    rtc.datetime((y,mo,d,wday,hh,mm,ss,0))

def now_local(tz=8):
    return time.localtime(time.time() + tz*3600)

# 1) 外部 RTC -> 系统 RTC
y,mo,d,hh,mm,ss,_ = chip.datetime()
set_system_rtc(y,mo,d,hh,mm,ss)
print("[RTC] DS1302 -> 系统时间:", now_local(8))

# 2) 若能上网，用 NTP 校时并回写 DS1302
def wifi(ssid, pwd, timeout=10):
    sta=network.WLAN(network.STA_IF); sta.active(True); sta.connect(ssid,pwd)
    t0=time.ticks_ms()
    while not sta.isconnected() and time.ticks_diff(time.ticks_ms(),t0)<timeout*1000:
        time.sleep_ms(200)
    return sta.isconnected()

if wifi("your-ssid","your-pass"):
    try:
        ntptime.settime()  # 设置为 UTC
        y,mo,d,hh,mm,ss,_,_ = time.gmtime()
        chip.set_datetime(y,mo,d,hh,mm,ss)  # 回写 DS1302 (UTC)
        print("[RTC] NTP 同步完成，已回写 DS1302")
    except Exception as e:
        print("[RTC] NTP 失败：", e)

print("[RTC] 本地(+8):", now_local(8))
```

## 7. 工程与实践要点

- 3.3V 供电优先：
  - 如果 DS1302 以 5V 供电，它的 IO 高电平≈5V，直接接 ESP32 可能损坏。仅当确认模块带限压/分压并安全时才用 5V；否则请用 3.3V 。
- 写保护与 CH 位：
  - 写时间前先 `write_protect(False)`；设置完成可再打开写保护。
  - 秒寄存器 bit7 为 CH（Clock Halt），务必清零才能“走时”。
- 禁用涓流充电：
  - 连接 CR2032（不可充电）时，一定写 0 到 0x90 关闭涓流，避免隐患。
- 星期与时区：
  - DS1302 的星期寄存器范围 1..7；显示时区通过软件偏移处理。
- 长期精度：
  - DS1302 精度一般，建议系统运行中定期 NTP 校时（如 12–24 小时一次），并回写 RTC。
- 深度睡眠：
  - 系统 RTC 在深度睡眠也继续计时；唤醒后建议仍以 DS1302/或 NTP 再校一次以保持一致。

## 8. 常见问题与排查

- 读写无响应：
  - CE/CLK/IO 接反；CE 未拉高；IO 未配置为出/入；检查地线与供电。
- 一直不走/秒不变：
  - CH 位未清；调用 `halt(False)` 或在设置秒寄存器时清 bit7。
- 写失败/时间写不进去：
  - 写保护未关（WP.bit7=1）；先 `write_protect(False)`。
- 时间跳变或错误：
  - BCD/十进制混用；务必使用 `_i2bcd/_bcd2i` 转换。
- ESP32 异常或引脚受损：
  - 在 5V 供电下直连 IO；改为 3.3V 供电或加电平转换。

## 9. 小结

本章实现了 DS1302 的从零驱动：三线半双工时序、寄存器读写、CH/写保护与涓流充电控制，并提供了与系统 RTC/NTP 的联动方案。工程上强烈建议以 3.3V 供电并关闭涓流充电（CR2032），在联网场景定期 NTP 校时，从而在低成本前提下获得足够可靠的实时时钟能力。
